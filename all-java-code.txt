package com.example.revisionSpringBoot.client;
import com.example.revisionSpringBoot.dto.AddressResponse;
import com.example.revisionSpringBoot.dto.ServiceCallResult;
import com.example.revisionSpringBoot.exception.DownstreamServiceException;
import com.example.revisionSpringBoot.exception.DownstreamServiceTimeoutException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Profile;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Component;
import org.springframework.web.client.ResourceAccessException;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientResponseException;
import java.net.URI;
import java.util.Collections;
import java.util.List;
@Component
@Profile("!test")
@Slf4j
public class AddressClientImpl implements AddressClient {
    private final RestClient restClient;
    public AddressClientImpl(
            RestClient.Builder restClientBuilder,
            @Value("${address.service.base-url}") String baseUrl) {
        this.restClient = restClientBuilder
                .baseUrl(baseUrl)
                .build();
    }
    @Override
    public ServiceCallResult<List<AddressResponse>> getAddressesByUserId(Long userId) {
        try {
            List<AddressResponse> addresses = restClient
                    .get()
                    .uri(uriBuilder -> {
                        URI uri = uriBuilder
                                .path("/api/addresses/user/{userId}")
                                .build(userId);
                        log.info("Calling address URL: {}", uri);
                        return uri;
                    })
                    .retrieve()
                    .body(new ParameterizedTypeReference<List<AddressResponse>>() {});
            if (addresses == null || addresses.isEmpty()) {
                return ServiceCallResult.noData(
                        "ADDRESS_SERVICE",
                        "No addresses found for this user"
                );
            }
            return ServiceCallResult.withData(addresses);
        } catch (RestClientResponseException e) {
            if (e.getStatusCode().equals(HttpStatus.NOT_FOUND)) {
                log.debug("No addresses found for userId={}", userId);
                return ServiceCallResult.noData(
                        "ADDRESS_SERVICE",
                        "No addresses found for this user"
                );
            }
            if (e.getStatusCode().is5xxServerError()) {
                throw new DownstreamServiceException(
                        "ADDRESS_SERVICE",
                        HttpStatus.BAD_GATEWAY,
                        "Address service returned HTTP "
                                + e.getStatusCode().value(),
                        e
                );
            }
            throw new DownstreamServiceException(
                    "ADDRESS_SERVICE",
                    HttpStatus.BAD_GATEWAY,
                    "Address service returned HTTP "
                            + e.getStatusCode().value(),
                    e
            );
        } catch (ResourceAccessException e) {
            throw new DownstreamServiceTimeoutException(
                    "ADDRESS_SERVICE",
                    "Address service is unavailable or timed out",
                    e
            );
        }
    }
}

import org.springframework.web.reactive.function.client.WebClientRequestException;
import org.springframework.web.reactive.function.client.WebClientResponseException;

import java.util.Collections;
import java.util.List;
@Component
@Slf4j
public class OrderClientImpl implements OrderClient {
    private final WebClient webClient;
    public OrderClientImpl(
            @Qualifier("orderWebClient") WebClient webClient) {
        this.webClient = webClient;
    }
    @Override
    public ServiceCallResult<List<OrderResponse>> getOrdersByUserId(Long userId) {
        try {
            List<OrderResponse> orders = webClient
                    .get()
                    .uri(uriBuilder -> {
                        URI uri = uriBuilder
                                .path("/api/orders/user/{userId}")
                                .build(userId);

                        return uri;

                    .retrieve()
                    .bodyToFlux(OrderResponse.class)
                    .collectList()
                    .block();

                return ServiceCallResult.noData(
                        "ORDER_SERVICE",
                        "No orders found for this user"
                );

            return ServiceCallResult.withData(orders);

                log.debug("No orders found for userId={}", userId);
                return ServiceCallResult.noData(
                        "ORDER_SERVICE",
                        "No orders found for this user"
                );
            }
            if (e.getStatusCode().is5xxServerError()) {
                log.error(
                        "Order service returned HTTP {}",
                        e.getStatusCode().value(),
                        e
                );
                throw new DownstreamServiceException(
                        "ORDER_SERVICE",
                        HttpStatus.BAD_GATEWAY,
                        "Order service returned HTTP "
                                + e.getStatusCode().value(),
                        e
                );
            }
            throw new DownstreamServiceException(
                    "ORDER_SERVICE",
                    HttpStatus.BAD_GATEWAY,
                    "Order service returned HTTP "
                            + e.getStatusCode().value(),
                    e
            );
        } catch (WebClientRequestException e) {
            log.error(
                    "Order service is unavailable or timed out for userId={}",
                    userId,
                    e
            );
            throw new DownstreamServiceTimeoutException(
                    "ORDER_SERVICE",
                    "Order service is unavailable or timed out",

package com.example.revisionSpringBoot.dto;
import lombok.AllArgsConstructor;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;
@Getter
@Setter
@AllArgsConstructor
@NoArgsConstructor
public class ServiceCallResult<T> {
    private T data;
    private boolean hasData;
    private PartialResponseWarning warning;
    public static <T> ServiceCallResult<T> withData(T data) {
        boolean hasData = data != null;
        if (data instanceof java.util.Collection<?> collection) {
            hasData = !collection.isEmpty();
        }
        return new ServiceCallResult<>(
                data,
                hasData,
                null
        );
    }
    public static <T> ServiceCallResult<T> noData(
            String serviceName,
            String message) {
        return new ServiceCallResult<>(
                null,
                false,
                new PartialResponseWarning(
                        serviceName,
                        message
                )
        );
    }
}

import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import java.time.LocalDateTime;
@RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(UserNotFoundException.class)
    public ResponseEntity<ErrorDetails> handleUserNotFound(
            UserNotFoundException ex,
            HttpServletRequest request) {
        ErrorDetails errorDetails = new ErrorDetails();
        errorDetails.setTimestamp(LocalDateTime.now());
        errorDetails.setMessage(ex.getMessage());
        errorDetails.setPath(request.getRequestURI());
        errorDetails.setStatus(HttpStatus.NOT_FOUND.value());
        errorDetails.setErrorcode("USER_NOT_FOUND");
        return ResponseEntity
                .status(HttpStatus.NOT_FOUND)
                .body(errorDetails);
    }
    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ErrorDetails> handleResourceNotFoundException(ResourceNotFoundException exception, WebRequest webRequest) {
        ErrorDetails errorDetails = new ErrorDetails(
                LocalDateTime.now(),
                exception.getMessage(),
                webRequest.getDescription(false),
                HttpStatus.NOT_FOUND.value(),
                "USER_NOT_FOUND"
        );
        return new ResponseEntity<>(errorDetails, HttpStatus.NOT_FOUND);
    }
    @ExceptionHandler(DownstreamServiceException.class)
    public ResponseEntity<ErrorDetails> handleDownstreamServiceException(
            DownstreamServiceException exception,
            HttpServletRequest request) {
        ErrorDetails errorDetails = new ErrorDetails();
        errorDetails.setTimestamp(LocalDateTime.now());
        errorDetails.setMessage(
                exception.getServiceName()
                        + " is currently unavailable"
        );
        errorDetails.setPath(request.getRequestURI());
        errorDetails.setStatus(
                HttpStatus.BAD_GATEWAY.value()
        );
        errorDetails.setErrorcode(
                "DOWNSTREAM_SERVICE_ERROR"
        );
        return ResponseEntity
                .status(HttpStatus.BAD_GATEWAY)
                .body(errorDetails);
    }

            DownstreamServiceTimeoutException exception,

        ErrorDetails errorDetails = new ErrorDetails();

        errorDetails.setMessage(
                exception.getServiceName()

        );
        errorDetails.setPath(request.getRequestURI());
        errorDetails.setStatus(

package com.example.revisionSpringBoot.service.impl;
import com.example.revisionSpringBoot.client.AddressClient;
import com.example.revisionSpringBoot.client.AddressClientImpl;
import com.example.revisionSpringBoot.client.OrderClient;
import com.example.revisionSpringBoot.dto.*;
import com.example.revisionSpringBoot.entity.User;
import com.example.revisionSpringBoot.exception.DownstreamServiceException;
import com.example.revisionSpringBoot.exception.DownstreamServiceTimeoutException;
import com.example.revisionSpringBoot.exception.ResourceNotFoundException;
import com.example.revisionSpringBoot.exception.UserNotFoundException;
import com.example.revisionSpringBoot.mapper.AutoUserMapper;
import com.example.revisionSpringBoot.repository.UserRepository;
import com.example.revisionSpringBoot.service.UserService;
import lombok.AllArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.*;
import java.util.stream.Collectors;
@Service
@AllArgsConstructor
@Slf4j
public class UserServiceImpl implements UserService {
    private final OrderClient orderClient;
    private final UserRepository userRepository;
    private final AutoUserMapper autoUserMapper;
    private final AddressClient addressClient;
    @Override
    public UserResponse createUser(UserRequest userRequest) {
        User user = autoUserMapper.mapToUserEntity(userRequest);
        User saveUser = userRepository.save(user);
        UserResponse saveUserResponse = autoUserMapper.mapToUserResponse(saveUser);
        return saveUserResponse;
    }
    @Override
    public UserResponse getUserById(int userId) {
        log.info("received user id: {}", userId);
        User user = userRepository.findById(userId)
                .orElseThrow(() ->
                        new UserNotFoundException(
                                "User not found with id: " + userId
                        ));
        UserResponse userResponse =
                autoUserMapper.mapToUserResponse(user);
        List<PartialResponseWarning> warnings =
                new ArrayList<>();
        ServiceCallResult<List<AddressResponse>> addressResult =
                addressClient.getAddressesByUserId((long) userId);
        userResponse.setAddresses(
                addressResult.getData() != null
                        ? addressResult.getData()
                        : Collections.emptyList()
        );
        if (!addressResult.isHasData()
                && addressResult.getWarning() != null) {
            warnings.add(addressResult.getWarning());
        }
        ServiceCallResult<List<OrderResponse>> orderResult =
                orderClient.getOrdersByUserId((long) userId);
        userResponse.setOrders(
                orderResult.getData() != null
                        ? orderResult.getData()
                        : Collections.emptyList()
        );
        if (!orderResult.isHasData()
                && orderResult.getWarning() != null) {
            warnings.add(orderResult.getWarning());
        }
        userResponse.setWarnings(warnings);
        return userResponse;
    }
    @Override
    public List<UserResponse> getAllUsers() {
        List<User> users = userRepository.findAll();
        return users.stream()
                .map(autoUserMapper::mapToUserResponse)
                .collect(Collectors.toList());
    }
    @Override
    public UserResponse updateUserById(int userId, UserRequest user) {
        User existingUser = userRepository.findById(userId).orElseThrow(() -> new UserNotFoundException("User not found with id: " + userId));
        log.info("in user service implementation : user id --> {}, name --> {} {}, email --> {}", existingUser.getId(), existingUser.getFirstName(), existingUser.getLastName(), existingUser.getEmail());
        existingUser.setFirstName(user.getFirstName());
        existingUser.setLastName(user.getLastName());
        existingUser.setEmail(user.getEmail());
        User updateUser = userRepository.save(existingUser);
        log.info("in user service implementation : user id --> {}, name --> {} {}, email --> {}", updateUser.getId(), updateUser.getFirstName(), updateUser.getLastName(), updateUser.getEmail());
        return autoUserMapper.mapToUserResponse(updateUser);
    }
    public void deleteUserById(int userId) {
        User user = userRepository.findById(userId)
                .orElseThrow(() ->
                        new UserNotFoundException("User not found with id: " + userId)
                );
        userRepository.delete(user);
    }
    @Override
    @Transactional
    public Map<String, Object> updateUserByEmail(String email, String firstName, String lastName) {
        User existingUser = userRepository.findByEmail(email);
        if (existingUser == null) {
            throw new ResourceNotFoundException("User", "email", email);
        }
        userRepository.updateUserByEmail(email, firstName, lastName);
        Map<String, Object> response = new HashMap<>();
        response.put("email", email);
        response.put("id", existingUser.getId());
        response.put("msg", "firstname - \"" + firstName + "\" and lastname - \"" + lastName + "\" updated");
        return response;
    }
